		.DW(@$(BUS.WIDTH))

		// {{{
		.i_clk(@$(SLAVE.BUS.CLOCK.WIRE)), .i_reset(@$(SLAVE.BUS.CLOCK.RESET)),
		@$(SLAVE.ANSIPORTLIST),
		.i_mon_cyc(@$(MONBUS)_cyc),
		.i_mon_stb(@$(MONBUS)_stb),
		.i_mon_we(@$(MONBUS)_we),
		.i_mon_sel(@$(MONBUS)_sel),
		.i_mon_stall(@$(MONBUS)_stall),
		.i_mon_ack(@$(MONBUS)_ack),
		.i_mon_err(@$(MONBUS)_err)
		// }}}
	);

	// }}}

			p_numcyc;
	unsigned	p_control;
} @$(DEVID);
